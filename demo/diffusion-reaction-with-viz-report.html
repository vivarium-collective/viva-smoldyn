Composite spec (YAML)

name: diffusion-reaction-with-viz
description: "Smoldyn diffusion-reaction demo with a wired Visualization Step. Each update produces a Plotly HTML figure of the running molecule-count trajectories, written to the 'viz_html' store."

requires:
  processes: [SmoldynProcess, SmoldynPlots, RAMEmitter]

parameters:
  interval:
    type: float
    default: 5.0
    description: "Process interval in simulation time units."
  dt:
    type: float
    default: 0.01
    description: "Smoldyn internal timestep."
  rate_ab:
    type: float
    default: 0.01
    description: "First-order reaction rate constant for A -> B."
  viz_title:
    type: string
    default: "Smoldyn diffusion-reaction"
    description: "Title displayed at the top of the Plotly figure."

state:
  smoldyn:
    _type: process
    address: "local:SmoldynProcess"
    config:
      dimensions: 2
      bounds:
        - [0, 100]
        - [0, 100]
      boundary_type: r
      dt: "${dt}"
      seed: 12345
      species:
        A:
          difc: 1.0
          count: 200
          color: red
          display_size: 3
        B:
          difc: 0.3
          count: 0
          color: blue
          display_size: 3
      reactions:
        - name: convert
          subs: [A]
          prds: [B]
          rate: "${rate_ab}"
    inputs: {}
    outputs:
      molecule_counts: ["stores", "molecule_counts"]
      time: ["stores", "time"]
    interval: "${interval}"
  stores:
    molecule_counts: {}
    time: 0.0
    viz_html: ""
  viz:
    _type: step
    address: "local:SmoldynPlots"
    config:
      title: "${viz_title}"
    inputs:
      molecule_counts: ["stores", "molecule_counts"]
      time: ["stores", "time"]
    outputs:
      html: ["stores", "viz_html"]
  emitter:
    _type: step
    address: "local:ram-emitter"
    config:
      emit:
        molecule_counts: "map[integer]"
        time: "float"
    inputs:
      molecule_counts: ["stores", "molecule_counts"]
      time: ["stores", "time"]
